// Live JSON with a small file cache (no hard-coded model data).
function liveJson(url, name, maxAgeMin) {
  const f = path.join(SCRATCH, 'live-' + name + '.json');
  try { if (Date.now() - fs.statSync(f).mtimeMs < maxAgeMin * 60e3) return JSON.parse(fs.readFileSync(f, 'utf8')); } catch {}
  try { const out = execFileSync('curl', ['-s', '-m', '20', url], { encoding: 'utf8', maxBuffer: 64 * 1024 * 1024 }); const j = JSON.parse(out); fs.writeFileSync(f, out); return j; } catch {}
  try { return JSON.parse(fs.readFileSync(f, 'utf8')); } catch { return null; }
}

// Artificial Analysis Intelligence Index v5 for a route id: match the base model name, prefer the row whose
// reasoning variant matches the route (-high/-max/-xhigh...), else the best variant AA lists for that base.
function aaScore(model) {
  if (!AA_ROWS) { try { AA_ROWS = JSON.parse(fs.readFileSync(path.join(SCRATCH, 'aa-scores.json'), 'utf8')).rows; } catch { AA_ROWS = []; } }
  const id = String(model).toLowerCase().split('/').pop().replace(/:.*$/, '').replace(/-free$|-preview$/, '').replace(/[._ ]+/g, '-');
  const effort = (id.match(/-(max|xhigh|high|medium|low|minimal)$/) || [])[1] || '';
  const base = id.replace(/-(max|xhigh|high|medium|low|minimal|thinking|agentic)$/g, '').replace(/-(\d{4}|\d{6})$/, '');
  const rows = AA_ROWS.filter((r) => r.base === base || r.base.startsWith(base + '-') && /^\d{4,}$/.test(r.base.slice(base.length + 1)));
  if (!rows.length) return null;
  const exact = rows.find((r) => effort && r.variant === effort);
  return exact || rows.reduce((b, r) => (r.ii > b.ii ? r : b));
}

// Scoreboard lookup (Controller priors: tools/ai-brain/data/model-priors-live.json via cli.js priors refresh).
let PRIORS;
function priorScore(model) {
  if (!PRIORS) { try { PRIORS = JSON.parse(fs.readFileSync('<home>/AI/shipde-platform/tools/ai-brain/data/model-priors-live.json', 'utf8')).ranking.filter((x) => x.score); } catch { PRIORS = []; } }
  const n = String(model).toLowerCase().split('/').pop().replace(/:.*$/, '').replace(/-(low|medium|high|xhigh|max|thinking)$/, '').replace(/[._ ]/g, '-');
  let best = null;
  for (const x of PRIORS) { const c = x.canonical; if (n === c || n.startsWith(c + '-')) { if (!best || c.length > best.canonical.length) best = x; } }
  // raw benchmark score, not the in-source percentile (normalizedScore made dots-3 preview 96.7 vs raw 64.4)
  return best ? (Number.isFinite(best.rawScore) ? best.rawScore : best.score) : null;
}
function family(model) {
  // Family = the model's creator in the AA v5 table; otherwise the vendor segment of the id (openrouter-style
  // "<vendor>/<model>"), otherwise the leading word of the model name. No hard-coded name list.
  const s = aaScore(model);
  if (s && s.creator) return String(s.creator).toLowerCase();
  const parts = String(model).toLowerCase().replace(/^ninerouter\//, '').split('/');
  if (parts.length >= 3) return parts[parts.length - 2];
  return (parts[parts.length - 1].match(/^[a-z]+/) || ['other'])[0];
}

// Live discovery catalogue for the Controller (scratchpad, not the committed store), refreshed every 30 min.
const DISCOVERY_DIR = path.join(__dirname, 'discovery');

function refreshDiscovery() {

// 1. Controller ranking (refreshes main first so the ranking code is current).
function controllerRanking(item, role) {
  try { execFileSync('git', ['-C', MAIN_TREE, 'fetch', '-q', 'origin']); execFileSync('git', ['-C', MAIN_TREE, 'checkout', '-q', '--detach', 'origin/main']); } catch {}
  let out = '';
  try {
    refreshDiscovery();
    out = execFileSync('node', ['tools/ai-brain/cli.js', 'dispatch', '--dry-run', '--item', item, '--role', role, '--discovery-dir', DISCOVERY_DIR], { cwd: MAIN_TREE, env: { ...process.env, ...ninerouterEnv() }, encoding: 'utf8', timeout: 400000, stdio: ['ignore', 'pipe', 'pipe'] });
  } catch (e) { out = String(e.stdout || ''); }
  const rows = [];
  for (const line of out.split(/\r?\n/)) {
    const m = line.match(/^\s+\d+\.\s+(\S+)\s+\(score:\s*(\d+|QUALITY_UNKNOWN)\)/);
    if (!m) continue;
    const parts = m[1].split('::');
    // 05/11: Controller 'opencode-direct' rows are direct-provider routes whose last part is already an opencode
    // '<provider>/<model>' id (cohere/..., amd-radeon/...): launch them as ordinary opencode writers.
    const kind = parts[0] === 'opencode-direct' ? 'opencode' : parts[0];
    const model = parts[parts.length - 1];
    const lane = kind === 'agy-pool' ? (parts[1].match(/agy\d+/) || [null])[0] : null;
    // Controller has no grade for this model: use the scoreboard's, or skip it.
    const score = m[2] === 'QUALITY_UNKNOWN' ? priorScore(model) : Number(m[2]);
    if (score == null && m[2] !== 'QUALITY_UNKNOWN') continue; // ungraded Controller rows stay: AA v5 scores them below
    rows.push({ origin: m[2] === 'QUALITY_UNKNOWN' ? 'controller+scoreboard' : 'controller', kind, lane, model, score, key: m[1] });
  }
  return rows;
}

// 2a. Scored models that answered the latest live probe (non-Controller routes).
function probedRoutes() {
  if (!fs.existsSync(PROBE_OUT)) return { rows: [], ageMin: Infinity };
  const ageMin = (Date.now() - fs.statSync(PROBE_OUT).mtimeMs) / 60000;
  const rows = [];
  for (const line of fs.readFileSync(PROBE_OUT, 'utf8').split(/\r?\n/)) {
    const m = line.match(/^OK\s+([\d.]+)\s+(\S+)\s+(\S+)\s+200\s+(\S+)/);
    if (!m) continue;
    const route = m[4];
    if (/nano|lite|mini\b/i.test(route)) continue; // probe maps these onto the full model's prior
    // opencode refuses models its config does not list ("Model not found"), even if the provider serves them.
    if (!opencodeHas(route)) continue;
    rows.push({ origin: 'probe', kind: 'opencode', lane: null, model: route, score: Number(m[1]), tier: m[2], key: route });
  }
  // agentrouter (via codex CLI) serves models the opencode probe cannot reach; reuse the best prior score of the same model family from the probe file.
  for (const m of arModels()) {
    const base = m.replace(/[^a-z0-9]/gi, '').toLowerCase();
    let best = null;
    for (const line of fs.readFileSync(PROBE_OUT, 'utf8').split(/\r?\n/)) {
      const x = line.match(/^\w+\s+([\d.]+)\s+\S+\s+(\S+)/); if (!x) continue;
      if (x[2].replace(/[^a-z0-9]/gi, '').toLowerCase().startsWith(base) && (best == null || Number(x[1]) > best)) best = Number(x[1]);
    }
    if (best != null) for (const acc of AR_ACCOUNTS) rows.push({ origin: 'agentrouter', kind: acc.kind, arEnv: acc.env, lane: null, model: m, score: best, key: `${acc.kind}::${m}` });
  }
  return { rows, ageMin };
}

// 2b. Live checks per launch kind.
function lanes() {
  let txt = '';
  try { txt = execFileSync('schtasks', ['/Query', '/FO', 'CSV'], { encoding: 'utf8' }); } catch {}
  const running = new Set();
  for (const l of txt.split(/\r?\n/)) { const m = l.match(/ShipDe-(agy\d+)","[^"]*","Running"/); if (m) running.add(m[1]); }
  // schtasks may report Ready while agy still works: treat a lane as busy for 90 min after launch unless its worktree already committed.
  try {
    for (const l of fs.readFileSync(`${LOGS}/route-log.jsonl`, 'utf8').trim().split(/\r?\n/)) {
      const r = JSON.parse(l); const m = r.choice && /^(agy\d+):/.exec(r.choice.label); if (!m || !r.launch) continue;
      if (Date.now() - Date.parse(r.at) > 90 * 60000) continue;
      let c = ''; try { c = execFileSync('git', ['-C', r.cwd, 'log', '-1', `--since=${r.at}`, '--format=%h'], { encoding: 'utf8' }).trim(); } catch {}
      // a lane whose result.json says the run finished after this launch is free again, commit or not
      let fin = 0; try { fin = Date.parse(JSON.parse(fs.readFileSync(`${AGY}/${m[1]}/result.json`, 'utf8').replace(/^﻿/, '')).finishedAt) || 0; } catch {}
      // the pipeline (or coordinator) ended this lane after the launch: result.json never gets finishedAt, but the lane is free
      try { fin = Math.max(fin, Number(fs.readFileSync(path.join(SCRATCH, 'pipeline-state', `ended-${m[1]}`), 'utf8').trim()) || 0); } catch {}
      if (!c && fin < Date.parse(r.at)) running.add(m[1]);
    }
  } catch {}
  return running;
}
function agyQuota(lane, model) {
  try {
    const q = JSON.parse(fs.readFileSync(`${AGY}/${lane}/quota.json`, 'utf8').replace(/^﻿/, ''));
    const a = Array.isArray(q) ? q : (q.groups || Object.values(q));
    // pick the group whose own model description mentions the most words of this model id (live quota.json text)
    const words = String(model || '').toLowerCase().split(/[^a-z]+/).filter((w) => w.length > 2);
    const hits = (x) => words.filter((w) => String(x.models || x.name || '').toLowerCase().includes(w)).length;
    const g = a.reduce((b, x) => (hits(x) > (b ? hits(b) : 0) ? x : b), null);
    const reset = g && g.weekly && g.weekly.resetAt && Date.parse(g.weekly.resetAt) < Date.now();
    const wk = g && g.weekly ? g.weekly.remaining : null; if (wk == null) return null; return { weekly: reset ? 1 : wk, five: g.fiveHour ? g.fiveHour.remaining : null, stale: !!reset };
  } catch { return null; }
}
// audit 03/11 B: DNS/socket errors are NETWORK_ERROR with a 5-min backoff, never "not logged in".
function agyLoggedIn(lane) {
  try { const p = `${AGY}/${lane}/err.txt`; const e = fs.readFileSync(p, 'utf8').slice(-2000);
    if (/no such host|i.o timeout|connection refused|ECONNRESET|ENOTFOUND|dial tcp/i.test(e)) return (Date.now() - fs.statSync(p).mtimeMs) > 5 * 60000 ? true : 'network';
    return !/Authentication required|Verify your account|authentication failed/i.test(e); } catch { return true; }
}

function liveCheck(c, running) {
  if (c.kind === 'agy-pool') {
    if (!c.lane) return 'no lane id';
    if (running.has(c.lane)) return 'lane busy';
    const li = agyLoggedIn(c.lane); if (li === 'network') return 'network error (retry in 5 min)'; if (!li) return 'lane not logged in';
    const q = agyQuota(c.lane, c.model);
    if (!q) return 'no quota data';
    if (q.weekly < MIN_WEEKLY) return `weekly quota ${q.weekly.toFixed(2)}`;
    return null;
  }
  // operator 2026-11-03: agentrouter Codex is reserved for review, so writers stay independent of the reviewer
  if (/^codex-ar[0-9]*$/.test(c.kind)) return 'reserved as reviewer';
  if (c.kind === 'codex') return codexOk() ? null : 'codex usage-limited or unreachable';
  if (c.kind === 'opencode') return null; // answered the live probe
  return `kind ${c.kind} not launchable from coordinator`;
}

// opencode retries rate-limit / budget errors silently: the run log stays empty and the writer looks hung for 20 min.
// The real error is only in opencode's own log, so read its tail and cool those providers.
function opencodeProviderErrors() {
  const out = {}; const rvo = revived();
  try {
    const p = path.join(process.env.USERPROFILE || '', '.local/share/opencode/log/opencode.log');
    const size = fs.statSync(p).size, len = Math.min(size, 8 * 1024 * 1024), b = Buffer.alloc(len);
    const fd = fs.openSync(p, 'r'); fs.readSync(fd, b, 0, len, size - len); fs.closeSync(fd);
    for (const l of b.toString('utf8').split(/\r?\n/)) {
      if (!/stream error|Too many requests|Free usage limit|gateway_timeout|Cannot connect to API/i.test(l)) continue;
      const t = Date.parse((/timestamp=(\S+)/.exec(l) || [])[1]); let prov = (/providerID=(\S+)/.exec(l) || [])[1];
      const mid = (/modelID=(\S+)/.exec(l) || [])[1];
      if (prov === 'ninerouter' && mid) prov = `ninerouter/${mid}`; // exact route, not all of 9router
      if (!t || !prov) continue;
      if ((rvo[prov] || 0) > t) continue; // worked again after this error
      const age = Date.now() - t;
      const daily = /Request too large|\[413\]|reached the limit|MONTHLY_REQUEST|max_tokens. must be|\[40[0-3]\]|Daily usage limit|Free usage limit|balance is too low|quota|insufficient|rate_limit_day/i.test(l), rate = /Too Many Requests|rate[ _]limit|\b429\b|gateway_timeout|Cannot connect to API/i.test(l);
      if ((daily && age < 8 * 3600000) || (rate && age < 3600000)) out[prov] = daily ? 'daily budget/balance exhausted' : 'rate limited';
    }
  } catch {}
  return out;
}

// opencode processes older than 2h are hung leftovers (every real run ends far sooner): they must not hold provider slots
function providerLoad() {
  const load = {};
  try {
    const o = execFileSync('powershell', ['-NoProfile', '-Command', "Get-CimInstance Win32_Process | Where-Object { $_.Name -eq 'opencode.exe' -and $_.CommandLine -like '*run -m*' -and $_.CreationDate -gt (Get-Date).AddHours(-2) } | ForEach-Object { $_.CommandLine }"], { encoding: 'utf8', timeout: 60000 });
    for (const line of o.split(/\r?\n/).filter(Boolean)) { const m = line.match(/run -m "?([^"\s/]+)\//); if (m) load[m[1]] = (load[m[1]] || 0) + 1; }
  } catch {}
  return load;
}

  for (const l of lines.slice(-300)) {

    if (r.item !== item || !r.launch || !r.choice) continue;
    const started = Date.parse(r.at);
    if (Date.now() - started < 20 * 60000) continue; // may still be running
    if (Date.now() - started > 12 * 3600000) continue; // old failures expire (lane overwrites, outages)
    let last = 0;
    try { last = Number(execFileSync('git', ['-c', 'safe.directory=*', '-C', r.cwd, 'log', '-1', '--format=%ct'], { encoding: 'utf8' }).trim()) * 1000; } catch {}
    // A run that died on quota/rate limit says nothing about the model's ability.
    let t = ''; try { t = fs.readFileSync(r.launch.log, 'utf8').slice(-4000); if (/result\.json$/.test(r.launch.log)) t += fs.readFileSync(r.launch.log.replace('result.json', 'err.txt'), 'utf8').slice(-3000); } catch {}
    if (/"state":\s*"quota"|RESOURCE_EXHAUSTED|usage limit|rolling 24-hour|free-usage-exhausted|concurrency limit|\[429\]/i.test(t)) continue;
    // Infra failures are not the model's fault: unknown model id, a hung CLI we killed, and Codex runs (the sandbox cannot commit; the coordinator commits their output afterwards). agy lanes run as other Windows users and cannot write our .git either: same rule.
    if (/Model not found|killed hung writer|Unexpected server error|rejected permission to use|^\s*Error: \[\d{3}\]/im.test(t) || /^(codex|agy[0-9]+:)/.test(r.choice.label)) continue;
    if (last < started) out.add(modelKey(r.choice.label));
  }
  return out;
}

// Models whose run on this item produced a commit (a full round) in the last 48h: the item coming back means that round failed review.
function roundsUsed(item) {
  const out = new Set(); let lines = [];
  try { lines = fs.readFileSync(`${LOGS}/route-log.jsonl`, 'utf8').trim().split(/\r?\n/); } catch { return out; }
  const runs = [];
  for (const l of lines.slice(-400)) { let r; try { r = JSON.parse(l); } catch { continue; } if (r.item === item && r.launch && r.choice && Date.now() - Date.parse(r.at) < 48 * 3600000) runs.push(r); }
  for (let i = 0; i < runs.length; i++) {
    const r = runs[i], end = i + 1 < runs.length ? Date.parse(runs[i + 1].at) : Date.now();
    let n = 0; try { n = Number(execFileSync('git', ['-c', 'safe.directory=*', '-C', r.cwd, 'rev-list', '--count', `--since=${r.at}`, `--until=${new Date(end).toISOString()}`, '--no-merges', 'HEAD'], { encoding: 'utf8' }).trim()); } catch {}
    if (n > 0) out.add(modelKey(r.choice.label));
  }
  return out;
}

// 3. Jev.
function askJev(question, options, evidence) {
  try {
    const out = execFileSync('node', [path.join(SCRATCH, 'ask-jev.js'), question, ...options], { cwd: REPO + '/.worktrees/p-agy-lane', encoding: 'utf8', timeout: 300000, env: { ...process.env, JEV_EVIDENCE: evidence } });
    const line = out.trim().split(/\r?\n/).pop();
    return JSON.parse(line);
  } catch (e) { return { outcome: 'ERROR', reason: String(e.message).slice(0, 120) }; }
}

// 4. Launch.
function launch(c, cwd, taskFile, item) {
  let prompt = fs.readFileSync(taskFile, 'utf8');
  // 05/11: a branch that conflicts with main is never reviewed, so its commits are never pushed. Conflict comes first, whatever the step says.
  let conflict = false;
  try {
    execFileSync('git', ['-c', 'safe.directory=*', 'fetch', '-q', 'origin', 'main'], { cwd, timeout: 120000, stdio: 'ignore' });
    execFileSync('git', ['-c', 'safe.directory=*', 'merge-tree', '--write-tree', 'HEAD', 'origin/main'], { cwd, timeout: 60000, stdio: 'ignore' });
  } catch (e) { conflict = e.status === 1; }
  if (conflict) prompt = '# DO THIS FIRST\n\nThis branch conflicts with origin/main, so review and push are blocked. Run `git fetch origin main`, merge origin/main, resolve every conflict keeping both sides\' intent, run the checks and commit the merge. Then do the task below.\n\n' + prompt;
  try { prompt += '\n\n' + fs.readFileSync(path.join(SCRATCH, `scope-${item}.txt`), 'utf8'); } catch {}
  const tag = `${item}-${Date.now()}`;
  if (c.kind === 'agy-pool') {
    fs.writeFileSync(`${AGY}/${c.lane}/job.json`, JSON.stringify({ cwd, model: c.model, accountId: c.lane, printTimeoutSec: 3600, prompt: prompt + CODEX_SANDBOX_NOTE }));
    execFileSync('schtasks', ['/Run', '/TN', `\\ShipDe\\ShipDe-${c.lane}`]);
    return { how: `schtasks ShipDe-${c.lane}`, log: `${AGY}/${c.lane}/result.json` };
  }
  const log = `${LOGS}/route-${tag}.log`;
  const fd = fs.openSync(log, 'w');
  let child;
  if (/^codex-ar[0-9]*$/.test(c.kind)) {
    fs.writeFileSync(path.join(cwd, '.oc-codex-task.md'), prompt + CODEX_SANDBOX_NOTE);
    child = spawn('bash', ['-lc', `codex exec --skip-git-repo-check -c 'model_providers.ar={name="agentrouter",base_url="https://agentrouter.org/v1",env_key="${c.arEnv || 'AGENTROUTER_API_KEY'}",wire_api="responses"}' -c model_provider=ar -c disable_response_storage=true -m "${c.model}" -s workspace-write "$(cat .oc-codex-task.md)" < /dev/null; rm -f .oc-codex-task.md`], { cwd, detached: true, stdio: ['ignore', fd, fd] });
  } else
  if (c.kind === 'codex') {
    fs.writeFileSync(path.join(cwd, '.oc-codex-task.md'), prompt + CODEX_SANDBOX_NOTE);
    child = spawn('bash', ['-lc', `codex exec -m "${c.model}" -c model_reasoning_effort=medium -s workspace-write "$(cat .oc-codex-task.md)" < /dev/null; rm -f .oc-codex-task.md`], { cwd, detached: true, stdio: ['ignore', fd, fd] });
  } else {
    const local = path.join(cwd, '.oc');
    fs.mkdirSync(local, { recursive: true });
    fs.writeFileSync(path.join(local, 'task.md'), prompt + CODEX_SANDBOX_NOTE);
    child = spawn('bash', ['-lc', `opencode run -m "${c.model}" "Do the task in .oc/task.md. Work in the current directory." -f .oc/task.md < /dev/null; node "${path.join(SCRATCH, 'oc-continue.js').split(path.sep).join('/')}" "${log}" "${c.model}" < /dev/null`], { cwd, detached: true, stdio: ['ignore', fd, fd] });
  }
  child.unref();
  return { how: `${c.kind} pid ${child.pid}`, log };
}

    const ls = fs.readFileSync(path.join(LOGS, 'route-log.jsonl'), 'utf8').trim().split(/\r?\n/);
    for (let i = ls.length - 1; i >= 0; i--) {
      let r; try { r = JSON.parse(ls[i]); } catch { continue; }
      if (r.item !== a.item || !r.launch || !r.launch.log) continue;
      if (/^\s*Error:\s/m.test(fs.readFileSync(r.launch.log, 'utf8').replace(/\x1b\[[0-9;]*m/g, ''))) lastFailed = r.choice.label;
      break;
    }
  } catch {}
  const lblOf = (c) => (c.lane ? `${c.lane}:${c.model}` : `${c.kind}:${c.model}`);
  const usable = pool.some((c) => lblOf(c) !== lastFailed) ? pool.filter((c) => lblOf(c) !== lastFailed) : pool;
  const seenM = new Set(); const top = [];
  // 05/11 operator: "gọi trong controller và jev". Controller candidates decide; probe rows are only the fallback when none of them is usable.
  const fromController = usable.filter((c) => String(c.origin || '').startsWith('controller'));
  notes.push(`selection pool: ${fromController.length ? 'controller ' + fromController.length : 'probe fallback, controller had none usable'}`);
  for (const c of fromController.length ? fromController : usable) { const k = modelKey(String(c.model).replace(/:free$|-free$/, '').replace(/-(agentic|thinking|thinking-agentic)$/, '')); if (seenM.has(k)) continue; seenM.add(k); top.push(c); if (top.length >= 5) break; }
  const label = (c) => (c.lane ? `${c.lane}:${c.model}` : `${c.kind}:${c.model}`);

  let choice = null;
  let jev = { outcome: 'SKIPPED' };
  if (top.length) {
    // 04/11 operator: ask Jev per task with the task itself, its history and each model's record on THIS item.
    let rounds = 0; try { rounds = (fs.readFileSync(path.join(a.cwd, '.oc', 'history.md'), 'utf8').match(/VERDICT: CHANGES_REQUIRED/g) || []).length; } catch {}
    const job = sizeM && Number(sizeM[1]) > 400
      ? `shrink PR from ${sizeM[1]} to <=400 added lines by deleting code/tests no acceptance criterion needs, then fix ${nFind} review finding(s). Mostly deletion plus small fixes; needs reliable multi-file tool use and exact instruction following`
      : `fix ${nFind} review finding(s) at the cited file:line, no refactor`;
    let evidence = `Job for ${a.item} (area: ${area || 'unknown'}): ${job}. Risk: ${risky ? 'HIGH (product tenant/auth/money/carrier code)' : 'normal (internal tooling)'}. Review rounds so far: ${rounds}. ` +
      `Candidates (AA Intelligence v5 percentile; record on this repo): ` + top.map((c) => `${label(c)} ${c.score}${c.aa ? ' (AA-scored)' : ' (not on AA, probe prior only)'}; ${repoHistory(c)}; this item: ${itemHistory(a.item, c)}`).join(' | ') + `. Which one most likely finishes in one round?`;
    // 04/11: Jev returns LOW_CONFIDENCE once evidence exceeds ~2KB (measured: 2000 DECIDED, 8000 UNDECIDED) — keep it short.
    if (evidence.length > 1800) evidence = evidence.slice(0, 1800);
    try { fs.writeFileSync(path.join(SCRATCH, "jev-last-evidence.txt"), evidence); } catch {}
    // 05/11 operator + audit #10: Jev rates the job, it never picks a route. Candidates are ranked by Controller score and
    // split evenly over the rubric's levels: hard takes the top, simple the lower end. Unrated or repeatedly refused jobs take the top.
    const ranked = [...top].sort((x, y) => (y.score || 0) - (x.score || 0));
    const li = RUBRIC.findIndex((r) => r[0] === level);
    const fromTop = li < 0 || roundsSoFar >= 3 ? 0 : RUBRIC.length - 1 - li;
    choice = ranked[Math.floor(fromTop * ranked.length / RUBRIC.length)];
    jev = { outcome: level ? 'RATED_' + level.toUpperCase() : 'UNRATED', confidence: jobJev && jobJev.confidence };
    // 05/11 operator: 9router is called through the Controller's combo, which fails over across its models and accounts by itself.
    try {
      const combo = JSON.parse(fs.readFileSync(path.join(MAIN_TREE, 'tools/ai-brain/sources.json'), 'utf8')).dispatch.nineRouterWriterCombo;
      // 06/11: off until a concurrency-limited head no longer queues writers: the combo's head waited forever instead of failing over.
      if (process.env.ROUTE_VIA_COMBO && combo && choice && choice.kind === 'opencode' && String(choice.model).startsWith('ninerouter/')) {
        notes.push(`9router via combo ${combo} (Controller pick ${choice.model})`);
        choice = { ...choice, model: 'ninerouter/' + combo };
      }
    } catch {}
  }

  if (a['dry-run']) for (const c of ctl) { const lb = `${c.kind}:${c.lane ? c.lane + ':' : ''}${c.model}`; const r = rejected.find(([x]) => x === c); console.log(`  ctl[${c.origin}] ${lb} score=${c.score} -> ${r ? 'REJECTED ' + r[1] : eligible.includes(c) ? 'eligible' : 'dropped'}`); }
  const record = {
    at: new Date().toISOString(), item: a.item, role, cwd: a.cwd, exclude,
    controllerCandidates: ctl.length, probeRoutes: probe.rows.length, notes,
    top: top.map((c) => ({ label: label(c), score: c.score, origin: c.origin })),
    rejected: rejected.slice(0, 12).map(([c, w]) => ({ label: label(c), score: c.score, why: w })),
    jev, choice: choice ? { label: label(choice), score: choice.score, kind: choice.kind } : null,
  };
  if (choice && !a['dry-run']) record.launch = launch(choice, a.cwd, stepTask || a.task, a.item);
  if (stepTask) record.step = path.relative(a.cwd, stepTask);
  fs.appendFileSync(`${LOGS}/route-log.jsonl`, JSON.stringify(record) + '\n');

  console.log(`${a.item}: ${choice ? label(choice) + ' (score ' + choice.score + ')' : 'NO ELIGIBLE WRITER'} | jev=${jev.outcome}${jev.choice ? ':' + jev.choice : ''}${record.launch ? ' | ' + record.launch.how : a['dry-run'] ? ' | dry-run' : ''}`);
  console.log('  top: ' + top.map((c) => `${label(c)}=${c.score}`).join(', '));
  console.log('  rejected: ' + rejected.filter(([c])=>c.kind!=='paseo').slice(0, 400).map(([c, w]) => `${label(c)}=${c.score}(${w})`).join(', ') + ` [+${rejected.filter(([c])=>c.kind==='paseo').length} paseo not launchable]`);
  if (a['dry-run']) console.log('  paseo: ' + rejected.filter(([c])=>c.kind==='paseo').map(([c]) => `${label(c)}=${c.score}`).join(', '));
  if (notes.length) console.log('  notes: ' + notes.join('; '));
})();

function opencodeHas(route) {
  if (!opencodeHas.cfg) { try { opencodeHas.cfg = JSON.parse(fs.readFileSync(path.join(process.env.USERPROFILE || process.env.HOME, '.config/opencode/opencode.json'), 'utf8')).provider || {}; } catch { opencodeHas.cfg = {}; } }
  const i = route.indexOf('/'); const p = opencodeHas.cfg[route.slice(0, i)];
  return !!(p && p.models && Object.prototype.hasOwnProperty.call(p.models, route.slice(i + 1)));
}

// Launches of the same model in the last 20 min (any item). Free-tier models burn an account per long run,
// so spreading work across models beats stacking it on one that then exhausts everywhere at once.
function recentModelLaunches() {
  const n = {}; let lines = [];
  try { lines = fs.readFileSync(`${LOGS}/route-log.jsonl`, 'utf8').trim().split(/\r?\n/).slice(-200); } catch { return n; }
  for (const l of lines) { let r; try { r = JSON.parse(l); } catch { continue; } if (!r.launch || !r.choice) continue;
    if (Date.now() - Date.parse(r.at) < 20 * 60000) { const k = modelKey(r.choice.label); n[k] = (n[k] || 0) + 1; } }
  return n;
}
