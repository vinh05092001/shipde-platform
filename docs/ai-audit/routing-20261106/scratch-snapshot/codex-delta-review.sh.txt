#!/bin/bash
# reviewer via agentrouter (key 1) while the ChatGPT Codex account is usage-limited (2026-11-03); set AR= to return to ChatGPT
: "${AR=}"  # default = ChatGPT review account in <lane>; AR=1 -> agentrouter
# 04/11 operator: <lane> Codex is in use by the operator; review via agentrouter until 04/11-20:41
[ -z "$AR" ] && [ $(date +%s) -lt 1793799673 ] && AR=1
# 05/11 08:35: <lane> ChatGPT Codex refresh token revoked (401 token_revoked) -> agentrouter until operator re-logs <lane> and sets <LANE>_CODEX_OK=1
[ -z "$AR" ] && [ -z "$<LANE>_CODEX_OK" ] && AR=1
# agentrouter keys from the User env (names only here; values never printed); rotate per review
for n in AGENTROUTER_API_KEY AGENTROUTER_API_KEY_2; do [ -z "$(printenv $n)" ] && export $n="$(powershell -NoProfile -Command "[Environment]::GetEnvironmentVariable('$n','User')" | tr -d '\r\n')"; done
ARKEYS=""; for n in AGENTROUTER_API_KEY AGENTROUTER_API_KEY_2; do [ -n "$(printenv $n)" ] && ARKEYS="$ARKEYS $n"; done
NK=$(echo $ARKEYS | wc -w); [ "$NK" -gt 0 ] || NK=1; ROT=<scratch>/ar-rot.txt
I=$(( $(cat $ROT 2>/dev/null || echo 0) % NK )); echo $((I+1)) > $ROT
ARORDER=$(echo $ARKEYS | tr ' ' '\n' | awk -v i=$I '{a[NR-1]=$0} END{for(k=0;k<NR;k++) print a[(k+i)%NR]}' | tr '\n' ' ')
ARK=$(echo $ARORDER | cut -d' ' -f1); [ -n "$ARK" ] || ARK=AGENTROUTER_API_KEY
# usage: codex-delta-review.sh <pr> <tag> <base-sha> <prev-review-file|-> <effort> "<focus>"  ; waits CI, reviews base..head delta with Codex sol; prints 3 lines
PR=$1; TAG=$2; BASE=$3; PREV=$4; EFF=$5; FOCUS=$6; R=<home>/AI/shipde-platform; L=$R/.worktrees/logs/night
cd $R; for i in $(seq 1 15); do r=$(gh pr view $PR --json statusCheckRollup --jq '[.statusCheckRollup[]|(.conclusion//"")]|join(",")'); echo "$r" | grep -qE '(^|,)(,|$)' && { sleep 60; continue; }; break; done
git fetch -q origin; H=$(gh pr view $PR --json headRefOid --jq .headRefOid)
echo "CI #$PR ${H:0:8}: $(gh pr view $PR --json statusCheckRollup --jq '[.statusCheckRollup[]|select(.conclusion!="SUCCESS" and .conclusion!="SKIPPED")|.name+"="+(.conclusion//.status)]|join(",")')"
# Gate: a red CI is a CHANGES_REQUIRED by itself; do not spend reviewer tokens on it.
FAILED=$(gh pr view $PR --json statusCheckRollup --jq '[.statusCheckRollup[]|select(.conclusion=="FAILURE" or .conclusion=="CANCELLED" or .conclusion=="TIMED_OUT")|.name]|unique|join(", ")')
if [ -n "$FAILED" ]; then printf 'VERDICT: CHANGES_REQUIRED\nReviewed SHA: %s\n\n1. **P1 — CI failing:** %s. Reproduce locally with the same commands as CI, fix, and re-run them green before committing.\n' "$H" "$FAILED" > $L/review-$TAG.txt; bash <scratch>/ci-excerpt.sh $PR >> $L/review-$TAG.txt 2>/dev/null; echo "$TAG exit=0 tokens=0 VERDICT: CHANGES_REQUIRED (CI gate: $FAILED) | open P0/P1: 1"; exit 0; fi
OWN=$(for f in $(git diff --name-only $BASE $H); do [ "$(git rev-parse -q --verify $H:$f 2>/dev/null)" = "$(git rev-parse -q --verify origin/main:$f 2>/dev/null)" ] || echo "$f"; done); FROMMAIN=$(( $(git diff --name-only $BASE $H | wc -l) - $(echo "$OWN" | grep -c .) )); if [ -n "$OWN" ]; then git diff $BASE $H -- $OWN > $L/delta-$TAG.patch; else : > $L/delta-$TAG.patch; fi; echo "delta $TAG: $(echo "$OWN" | grep -c .) own files, $FROMMAIN identical-to-main files excluded, $(wc -c < $L/delta-$TAG.patch) bytes"
{ echo "You are the independent reviewer (fresh task) for PR #$PR. Do NOT run commands or read files; everything is below. One reply. Focus: $FOCUS. Check each open P0/P1 of the previous review is truly fixed in the delta (cite file:line) and list NEW P0/P1. Output: line 1 VERDICT: PASS or VERDICT: CHANGES_REQUIRED, line 2 Reviewed SHA: $H, then numbered findings. P2 alone does not block."; cat <scratch>/review-format.md; WID=$(gh pr view $PR --json title --jq .title | grep -oE "[A-Z]+-[A-Z]+-[0-9]+" | head -1); echo "=== WORK ITEM $WID (goal + acceptance) ==="; git -C <home>/AI/shipde-platform show "$H:docs/product-spec/work-items/$WID.md" 2>/dev/null | head -80; echo; [ "$PREV" != "-" ] && { echo "=== PREVIOUS REVIEW ==="; cat $L/$PREV; echo; }; echo "=== GITHUB CI (gh pr view statusCheckRollup, exact head $H) ==="; gh pr view $PR --json statusCheckRollup --jq ".statusCheckRollup[]|\"\(.name)=\(.conclusion//.status) commit=$H\""; echo "NOTE: $FROMMAIN changed files whose content at head is byte-identical to origin/main (brought in by merging main) are excluded; they are main code, not this PR. Review only the files below."; echo "=== DELTA $BASE -> $H ==="; cat $L/delta-$TAG.patch; WT=$(grep -E "^[^#][^|]*|[^|]*|[^|]*|$PR|" <scratch>/items.txt | head -1 | cut -d"|" -f1); DL=$R/.worktrees/$WT/.oc/decisions.md; [ -n "$WT" ] && [ -s "$DL" ] && { echo "=== DECISION LEDGER (settled across ALL earlier rounds; a finding that restates an entry here in other words is NOT new: report it only under prior findings as NOT FIXED if the delta truly reverted it) ==="; tail -c 12000 "$DL"; }; } > $L/rvprompt-$TAG.txt
# 05/11 operator: reviewer family must differ from the writer (combo "vinh" writes with gpt-4.1); picked by AA score from live routes
# 05/11: one direct chat call for 9router reviewers (r9.js, streaming); opencode only for other providers
rv_one() { if [ "${1#ninerouter/}" != "$1" ]; then (cd $L && node <scratch>/r9.js "${1#ninerouter/}" rvprompt-$TAG.txt review-raw-$TAG.txt >> rvlog-$TAG.log 2>&1; tr -d "\r" < review-raw-$TAG.txt | awk "/^VERDICT:/{f=1} f" > review-$TAG.txt); else cd $L && D=$L/oc-$TAG; mkdir -p $D && cp rvprompt-$TAG.txt $D/prompt.md && (cd $D && timeout 1800 opencode run -m "$OC" "Follow prompt.md exactly. Do not use any tools. Output only the review." -f prompt.md < /dev/null 2>&1 | sed "s/[[0-9;]*m//g" | tee -a $L/rvraw-$TAG.log | awk "/^VERDICT:/{f=1} f" > $L/review-$TAG.txt) >> rvlog-$TAG.log 2>&1; fi; printf '{"model":"%s","ok":%s,"at":"%s","tag":"%s"}\n' "$1" "$([ -s $L/review-$TAG.txt ] && echo true || echo false)" "$(date -u +%Y-%m-%dT%H:%M:%SZ)" "$TAG" >> <scratch>/reviewer-results.jsonl; }
if [ -z "$OC" ] && [ -z "$USE_CODEX_REVIEW" ]; then OCLIST=$(node <scratch>/reviewer-pick.js $TAG 2>/dev/null); OC=$(echo $OCLIST | cut -d" " -f1); fi
cd $L && if [ -n "$OCLIST" ] && [ $(echo $OCLIST | wc -w) -ge 2 ]; then node <scratch>/review-many.js $TAG $OCLIST >> rvlog-$TAG.log 2>&1; OCLIST=; elif [ -n "$OC" ]; then rv_one "$OC"; elif [ -n "$R9" ]; then node <scratch>/r9.js $R9 rvprompt-$TAG.txt review-$TAG.txt > rvlog-$TAG.log 2>&1; elif [ -z "$AR" ]; then bash <scratch>/codex-<lane>.sh rvprompt-$TAG.txt review-$TAG.txt ${RVM:-gpt-6.1-sol} $EFF > rvlog-$TAG.log 2>&1; else timeout 1800 codex exec --skip-git-repo-check -s read-only ${AR:+-c 'model_providers.ar={name="agentrouter",base_url="https://agentrouter.org/v1",env_key="'"$ARK"'",wire_api="responses"}' -c model_provider=ar} -m ${RVM:-gpt-6.1-sol} -c model_reasoning_effort=$EFF -o review-$TAG.txt - < rvprompt-$TAG.txt > rvlog-$TAG.log 2>&1; fi
if [ -n "$OCLIST" ] && [ ! -s $L/review-$TAG.txt ]; then for OC in $(echo $OCLIST | cut -d" " -f2-); do echo "reviewer fallback -> $OC" >> $L/rvlog-$TAG.log; rv_one "$OC"; [ -s $L/review-$TAG.txt ] && break; done; fi
echo "reviewer: ${OC:-codex}" >> $L/rvlog-$TAG.log
if [ -n "$AR" ] && grep -qiE "quota is not enough|insufficient_quota" $L/rvlog-$TAG.log 2>/dev/null; then for ARK in $(echo $ARORDER | cut -d' ' -f2-); do echo "agentrouter key rotated to $ARK" >> $L/rvlog-$TAG.log; (cd $L && timeout 1800 codex exec --skip-git-repo-check -s read-only -c "model_providers.ar={name=\"agentrouter\",base_url=\"https://agentrouter.org/v1\",env_key=\"$ARK\",wire_api=\"responses\"}" -c model_provider=ar -m ${RVM:-gpt-6.1-sol} -c model_reasoning_effort=$EFF -o review-$TAG.txt - < rvprompt-$TAG.txt >> rvlog-$TAG.log 2>&1); grep -qiE "quota is not enough|insufficient_quota" $L/rvlog-$TAG.log || break; done; fi
echo "$TAG exit=$? tokens=$(grep -A1 'tokens used' rvlog-$TAG.log | tail -1) $(head -1 review-$TAG.txt) | open P0/P1: $(grep -E 'P0|P1' review-$TAG.txt | grep -viE 'closed|fixed' | wc -l)"
