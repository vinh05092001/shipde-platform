test('DNS rebinding between check and connect is refused', async () => {
  const root = mkdtempSync(join(tmpdir(), 'shipde-research-rebinding-'));
  const privateRef = join(root, 'private-listener.mjs');
  const privateHits = join(root, 'private-hits');
  const marker = join(root, 'cli-calls');
  try {
    await startListener(privateRef);
    const privateProc = spawn(process.execPath, [privateRef, privateHits, 'private data', '-']);
    const privatePort = Number(
      await new Promise((resolvePort, rejectPort) => {
        privateProc.stdout.once('data', (chunk) => resolvePort(String(chunk).trim()));
        privateProc.once('exit', (code) => rejectPort(new Error(`listener exited ${code}`)));
      })
    );
    try {
      // The host answers public on the first lookup and private on every later
      // one, so any connect that re-looks-up instead of using the validated
      // address would reach the private listener.
      let callCount = 0;
      const rebindingResolver = async () => {
        callCount++;
        return callCount === 1 ? ['93.184.216.34'] : ['127.0.0.1'];
      };

      const result = await runResearch(
        ['web', 'read', `http://rebind.example.com:${privatePort}/secret`],
        {
          binary: process.execPath,
          prefixArgs: [fixture, 'retrieve-target', marker],
          resolveHost: rebindingResolver,
        }
      );

      assert.notEqual(result.code, 0, 'Must be refused');
      assert.equal(existsSync(privateHits), false, 'Private listener must receive zero requests');
      assert.doesNotMatch(result.stdout + result.stderr, /private data/);
    } finally {
      privateProc.kill();
    }
  } finally {
    rmSync(root, { recursive: true, force: true });
  }
});

// Separate-process listeners so an assertion of zero requests proves real
// transport behavior instead of a blocked in-process event loop.
// In-process public listener reached through the pinned (loopback) address the
// test resolver returns, so no external network is needed. It records every
// hit and can answer with a late redirect.
async function startPublicServer(redirectTo) {
  const { createServer } = await import('node:http');
  const hits = [];
  const server = createServer((req, res) => {
    hits.push(req.url);
    if (redirectTo !== undefined && hits.length >= redirectTo.after) {
      res.writeHead(302, { location: redirectTo.location });
      res.end();
      return;
    }
    res.writeHead(200, { 'content-type': 'text/plain' });
    res.end(redirectTo?.body ?? 'public content');
  });
  await new Promise((resolveListen) => server.listen(0, '127.0.0.1', resolveListen));
  return { server, hits, port: server.address().port };
}

// A separate-process listener for the private destination so that a zero-hit
// assertion is proof of transport behavior, not a blocked event loop.
async function startPrivateProbe(root) {
  const ref = join(root, 'private-listener.mjs');
  const hits = join(root, 'private-hits');
  writeFileSync(
    ref,
    `#!/usr/bin/env node
import { appendFileSync } from 'node:fs';
import { createServer } from 'node:http';
const [counter] = process.argv.slice(2);
const server = createServer((req, res) => {
  appendFileSync(counter, req.url + '\\n');
  res.end('private content');
});
server.listen(0, '127.0.0.1', () => {
  process.stdout.write(String(server.address().port) + '\\n');
});
`
  );
  const proc = spawn(process.execPath, [ref, hits]);
  const port = await new Promise((resolvePort, rejectPort) => {
    proc.stdout.once('data', (chunk) => resolvePort(Number(String(chunk).trim())));
    proc.once('exit', (code) => rejectPort(new Error(`listener exited ${code}`)));
  });
  return { proc, hits, port };
}

test('redirect to private address during retrieval is refused', async () => {
  const root = mkdtempSync(join(tmpdir(), 'shipde-research-redirect-refusal-'));
  const publicRef = join(root, 'public-listener.mjs');
  const publicHits = join(root, 'public-hits');
  const privateRef = join(root, 'private-listener.mjs');
  const privateHits = join(root, 'private-hits');
  const marker = join(root, 'cli-calls');
  try {
    await startListener(publicRef);
    await startListener(privateRef);
    const privateProc = spawn(process.execPath, [privateRef, privateHits, 'private content', '-']);
    const privatePort = Number(
      await new Promise((resolvePort, rejectPort) => {
        privateProc.stdout.once('data', (chunk) => resolvePort(String(chunk).trim()));
        privateProc.once('exit', (code) => rejectPort(new Error(`listener exited ${code}`)));
      })
    );
    const publicProc = spawn(process.execPath, [
      publicRef,
      publicHits,
      'public content',
      `http://127.0.0.1:${privatePort}/secret`,
    ]);
    const publicPort = Number(
      await new Promise((resolvePort, rejectPort) => {
        publicProc.stdout.once('data', (chunk) => resolvePort(String(chunk).trim()));
        publicProc.once('exit', (code) => rejectPort(new Error(`listener exited ${code}`)));
      })
    );
    try {
      // The default transport performs the retrieval in-process; the public
      // server answers 200 and only the retrieval connection meets the 302 to
      // loopback, so a preflight alone cannot catch it.
      const result = await runResearch(
        ['web', 'read', `http://docs.example.com:${publicPort}/doc`],
        {
          binary: process.execPath,
          prefixArgs: [fixture, 'retrieve-target', marker],
          resolveHost: defaultTestResolver,
        }
      );

      assert.notEqual(result.code, 0, 'Must be refused');
      assert.equal(existsSync(privateHits), false, 'Private listener must receive zero requests');
      assert.equal(existsSync(marker), false, 'CLI must not be launched for retrieval');
      assert.doesNotMatch(result.stdout + result.stderr, /private content/);
      assert.ok(readFileSync(publicHits, 'utf8').length > 0, 'public destination was retrieved');
    } finally {
      publicProc.kill();
      privateProc.kill();
    }
  } finally {
    rmSync(root, { recursive: true, force: true });
  }
});
